## Tile-grid Map

A grid cartogram that needs no map geometry.

```jsx live
const data = [
  { id: "WA", row: 0, col: 0, value: 62 }, { id: "MN", row: 0, col: 3, value: 40 }, { id: "NY", row: 0, col: 7, value: 88 },
  { id: "CA", row: 1, col: 0, value: 95 }, { id: "IL", row: 1, col: 4, value: 71 }, { id: "PA", row: 1, col: 7, value: 66 },
  { id: "CO", row: 2, col: 3, value: 34 }, { id: "VA", row: 2, col: 7, value: 58 },
  { id: "TX", row: 3, col: 3, value: 82 }, { id: "GA", row: 3, col: 6, value: 49 }, { id: "FL", row: 4, col: 7, value: 77 },
];

const App = () => (
  <ChartThemeProvider mode="light">
    <ResponsiveContainer height={320}>
      {(width, height) => <TileGridMap width={width} height={height} data={data} />}
    </ResponsiveContainer>
  </ChartThemeProvider>
);
```

### Import

```tsx
import { TileGridMap, ChartThemeProvider, ResponsiveContainer } from "@commercetools/nimbus-viz";
```
