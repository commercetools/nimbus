## Waffle Chart

A grid of squares that reads share by counting cells.

```jsx live
const data = [
  { category: "Web", value: 4200 },
  { category: "Mobile", value: 3100 },
  { category: "Retail", value: 2400 },
  { category: "Partner", value: 1800 },
  { category: "Email", value: 1200 },
];

const App = () => (
  <ChartThemeProvider mode="light">
    <ResponsiveContainer height={280}>
      {(width, height) => <WaffleChart width={width} height={height} data={data} />}
    </ResponsiveContainer>
  </ChartThemeProvider>
);
```

### Import

```tsx
import { WaffleChart, ChartThemeProvider, ResponsiveContainer } from "@commercetools/nimbus-viz";
```
